br_ere_subject_events.1 = { 
	type = country_event
	title = br_ere_resistance_events.201.title
	desc = br_ere_resistance_events.201.desc
	outcome = neutral
	trigger = {
		liberty_desire > 50
		OR = {
			is_subject_type = pronoia
			is_subject_type = br_thema_subject
		}
		OR = {
			current_age = age_5_absolutism
			current_age = age_6_revolutions
		}
	}

	illustration_tags = {
        10 = regular
        10 = exterior
    }

	immediate = {
		event_illustration_poptype_effect = { foreground = pop_type:soldiers background = pop_type:soldiers }
		save_scope_as = from
		create_character = {
			trait_category = general
			save_scope_as = new_general_scope
			create_in_limbo = yes
		}
	}

	option = {
		name = br_ere_resistance_events.201.a
		scope:new_general_scope = { move_country = root }

		overlord = {
			trigger_event_silently = { id = br_ere_resistance_events.202 }
		}
	}
	option = {
		name = br_ere_resistance_events.201.b
		add_liberty_desire = liberty_desire_mild_plus
		overlord = {
			trigger_event_silently = { id = br_ere_resistance_events.203 }
		}
		kill_character = {
			target = scope:new_general_scope
			reason = execution
			killer = ruler
		}
	}
}

br_ere_subject_events.50 = { #Katapanate of Antioch
	type = country_event
	title = br_ere_subject_events.50.title
	desc = br_ere_subject_events.50.desc
	outcome = neutral
	fire_only_once = yes

	dynamic_historical_event = {
		tag = AOI
		from = 1345.1.1
		to = 1800.1.1
		monthly_chance = 10
	}

	illustration_tags = {
		10 = happy
		10 = interior
	}

	immediate = {
		event_illustration_poptype_effect = { foreground = pop_type:soldiers background = pop_type:soldiers }

		ruler ?= {
			save_scope_as = target_ruler
		}
		international_organization:roman_world.leader_country = {
			save_scope_as = target_emperor_country
		}
		international_organization:roman_world.leader_country.ruler = {
			save_scope_as = target_emperor
		}
		root = {
			save_scope_as = origin_country
		}
	}

	trigger = {
		has_ruler = yes
		is_member_of_international_organization = international_organization:roman_world
		NOT = {
			is_leader_of_international_organization = international_organization:roman_world
		}
		international_organization:roman_world = {
			international_organization_has_leader = yes
			leader_country = {
				has_ruler = yes
			}
		}
		owns_or_non_sovereign_subject_owns = location:antioch
		owns_or_non_sovereign_subject_owns = location:tarsus
		owns_or_non_sovereign_subject_owns = location:nicosia_cyp
	}

	option = {
		name = br_ere_subject_events.50.a
		historical_option = yes

		scope:target_emperor_country = {
			trigger_event_non_silently = { #Ask the Emperor for the title
				id = br_ere_subject_events.51
			}
		}
	}
}
