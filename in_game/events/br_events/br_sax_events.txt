br_sax_events.7 = { #The Rise of Hanover
	type = country_event
	title = br_sax_events.7.title
	desc = br_sax_events.7.desc
	outcome = neutral

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SAX
		from = 1371.1.1
		to = 1520.1.1
		monthly_chance = 10
	}

	trigger = {
		owns = location:hanover
		location:hanover = {
			population >= 25
			NOR = {
				location_rank = location_rank:town
				location_rank = location_rank:city
				location_rank = location_rank:megalopolis
			}
		}
	}
illustration_tags = {
        10 = regular
        10 = exterior
    }

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:burghers_estate }
	}

	option = {
		name = br_sax_events.3.a
		historical_option = yes

		trigger = {
			NOT = { capital = location:hanover }
		}

		change_gold_effect = { scale = -5.00 }

		if = {
			limit = {
				location:hanover = {
					is_capital = no
				}
			}
			set_capital = location:hanover
		}

		location:hanover = {
			if = {
				limit = {
					location_rank = location_rank:rural_settlement
				}
				change_location_rank_effect = { location_rank = location_rank:town }
			}
			change_development  = development_severe_bonus
		}
	}

	option = {
		name = br_sax_events.3.b
		change_gold_effect = { scale = -3.00 }
		location:dresden = {
			if = {
				limit = {
					location_rank = location_rank:rural_settlement
				}
				change_location_rank_effect = { location_rank = location_rank:town }
			}
			change_development  = development_severe_bonus
		}
	}
}

#Archduchy of Saxony Event Chain
br_sax_events.1000. = { #Archduchy of Saxony
	hide_portraits = yes
	type = country_event
	title = br_sax_events.1000.title
	desc = br_sax_events.1000.desc
	outcome = neutral

	fire_only_once = yes

	dynamic_historical_event = {
		tag = SAX
		from = 1340.1.1
		to = 1800.1.1
		monthly_chance = 2
	}

	illustration_tags = {
		10 = happy
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }

		ruler ?= {
			save_scope_as = target_ruler
		}
		international_organization:hre.leader_country = {
			save_scope_as = target_emperor_country
		}
		international_organization:hre.leader_country.ruler = {
			save_scope_as = target_emperor
		}
		root = {
			save_scope_as = origin_country
		}
	}

	trigger = {
		has_ruler = yes
		is_member_of_international_organization = international_organization:hre
		NOT = {
			is_leader_of_international_organization = international_organization:hre
		}
		international_organization:hre = {
			international_organization_has_leader = yes
			leader_country = {
				has_ruler = yes
			}
		}
		owns = location:berlin
		owns = location:hamburg
		owns = location:magdeburg
	}

	option = {
		name = br_sax_events.1000.a
		historical_option = yes

		scope:target_emperor_country = {
			trigger_event_non_silently = { #Ask the Emperor for the title
				id = br_sax_events.1001
			}
		}
	}
}
