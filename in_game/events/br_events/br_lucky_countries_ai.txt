br_lucky_countries_ai.1 = {
	type = country_event
	outcome = neutral
	title = empty_text
	desc = empty_text
	hidden = yes 

	dynamic_historical_event = {
        tag = SER
		tag = TUN
		tag = MZF
        from = 1337.1.1
        to = 1338.1.1
        monthly_chance = 100
    }

	trigger = {
		is_ai = yes
	}

	immediate = { #33% x100 years
		add_country_modifier = {
			modifier = br_lucky_country_buff
			years = 100
			mode = add_and_extend
			size = 0.33
		}
	}

}

br_lucky_countries_ai.2 = {
	type = country_event
	outcome = neutral
	title = empty_text
	desc = empty_text
	hidden = yes
	fire_only_once = yes

	dynamic_historical_event = {
        tag = LIT
        from = 1337.1.1
        to = 1338.1.1
        monthly_chance = 100
    }

	trigger = {
		is_ai = yes
		c:KIE ?= {
			is_ai = yes
		}
		c:CHR ?= {
			is_ai = yes
		}
	}

	immediate = {
		add_country_modifier = {
			modifier = br_lucky_country_buff
			years = 100
			mode = add_and_extend
			size = 0.33
		}
		c:KIE ?= {
			make_subject_of = {
				target = c:LIT
				type = subject_type:vassal
			}
		}
		c:CHR ?= {
			make_subject_of = {
				target = c:LIT
				type = subject_type:vassal
			}
		}
		every_subject = {
			grant_estate_privilege = estate_privilege:br_russian_subject_privilege
		}
	}	

}
