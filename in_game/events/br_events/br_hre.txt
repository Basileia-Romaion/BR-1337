br_hre_events.7 = { #Imperial civil war begun
    type = country_event
    title = br_hre_events.7.title
    desc = br_hre_events.7.desc
    outcome = neutral
    image = "gfx/interface/illustrations/disaster/curse_of_stefan_uros_iii.dds"
	fire_only_once = yes

	illustration_tags = {
		10 = regular
		10 = interior
	}

    option = {
        name = br_hre_events.7.a

        international_organization:prince_faction = {
       leader_country = {
        declare_war_with_cb = {
            target = c:LOT
            type = casus_belli:cb_imperial_civil_war
           }
         }
       }

        # Retrieve the war
        c:LOT = {
            random_current_war = {
                limit = {
                    attacker_leader = {
                        is_leader_of_international_organization = international_organization:prince_faction
                    }
                    defender_leader = c:LOT
                }
                save_scope_as = imperial_civil_war
            }
        }

        # Centralizers join the emperor
        every_international_organizations_member_of = {
            limit = {
                international_organization_type = international_organization_type:emperor_faction
            }
            every_international_organization_member = {
                limit = {
                    NOT = { tag = LOT }
                }
                join_war_as_defender = {
                    war = scope:imperial_civil_war
                }
            }
        }

        every_international_organizations_member_of = {
            limit = {
                international_organization_type = international_organization_type:prince_faction
            }
            every_international_organization_member = {
                limit = {
                    NOT = { tag = UBV }
                    NOT = { tag = LOT }
                }
                join_war_as_attacker = {
                    war = scope:imperial_civil_war
                }
            }
        }

		if = {
			limit = {
				situation:guelphs_and_ghibellines = {
					situation_is_active = yes
				}
			}
			end_situation = {
				situation = situation:guelphs_and_ghibellines
				outcome = "guelphs_and_ghibellines_factional_deadlock"
			}
		}

        end_situation = {
            situation = situation:br_hre_civil_war
            outcome = "br_hre_civil_war_war_declared"
        }
    }
}
