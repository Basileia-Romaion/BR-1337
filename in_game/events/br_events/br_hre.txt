#hidden_effect = {

namespace = br_hre_events

#Hidden HRE starting event is to balance the massive negative opinion all members have due to the mods starting laws. And it's a hidden event
br_hre_events.1 = {
	type = country_event
	outcome = neutral
	title = empty_text
	desc = empty_text
	hidden = yes
	fire_only_once = yes
	immediate = {
		custom_tooltip = {
			text = br_hre_events.1.att
			international_organization:hre ?= {
				every_international_organization_member = {
					limit = { 
						NOT = { is_leader_of_international_organization = root }
					}
					add_opinion = {
						target = c:LOT
						modifier = br_hre_start_opinion
					}
				}
			}
		}
	}
}
