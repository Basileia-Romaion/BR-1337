# Form Elysium
elysium_event.1 = {
	type = country_event
	title = elysium_event.1.title
	desc = elysium_event.1.desc
	outcome = positive

	illustration_tags = {
		10 = armed
		10 = exterior
	}

	dynamic_historical_event = {
        tag = ELY
        from = 1337.1.1
        to = 1800.1.1
        monthly_chance = 100
    }

	trigger = {
		#Should trigger on country formation.
	}

	immediate = {
		create_character = {
			trait_category = artist
			artist = scientist
			estate = estate_type:burghers_estate
			artist_skill = { 0.4 0.9 }
			culture = root.culture
			religion = root.religion
			min_age = 30
			create_in_limbo = yes
			save_scope_as = the_scientist
		}
		location:anacostia = {
			save_scope_as = target_location
		}
		capital = {
			save_scope_as = current_capital
		}
	}

	option = {
		name = elysium_event.1.a
		historical_option = yes
		high_reward_option = yes

		add_prestige = prestige_extreme_bonus
		change_culture = culture:elysian_culture
		scope:current_capital = {
			add_location_modifier = {
				modifier = influence_of_the_old_capital_modifier
				years = 100
				mode = add_and_extend
			}
		}
		move_capital_event_effect = {
			from = scope:current_capital
			to = scope:target_location
		}
		#hidden_effect = {
 			scope:target_location = {
 				rename_location = Elysium
				change_location_rank = location_rank:city
				change_raw_material = goods:silk
				add_location_modifier = {
					modifier = birth_of_a_new_city_adm
					years = 25
					mode = add_and_extend
				}
				add_location_modifier = {
					modifier = br_new_world_elysian_capital_2
					years = -1
					mode = add_and_extend
				}
				# figure out how to set vegetation = farmland
 			}
			add_country_modifier = {
				modifier = br_elysium_formed
				years = 10
				mode = add_and_extend
			}
			capital = {
                change_institution_progress = {
                    type = institution:meritocracy
                    value = 100
                }
            }
			every_owned_location = {
				every_pop = {
					limit = {
						OR = {
							culture = culture:aureliani
							culture = culture:photianoi
						}
					}
					change_pop_culture = culture:elysian_culture
				}
			}
 		#}
		hidden_effect = { #Goal is to split these two effects off so they are a more drawn out process via multiple events that happen after Elysium formation.
			scope:target_location = {
				change_development = development_extreme_bonus
				change_prosperity = prosperity_radical_bonus
				every_pop = {
					limit = {
						NOT = { culture = { is_accepted_in = root } }
					}
					change_pop_culture = root.culture
				}
 			}
			every_owned_location = {
				every_pop = {
					limit = {
						OR = {
							culture = { has_culture_group = culture_group:powhatan_group }
							culture = { has_culture_group = culture_group:lenape_group }
							culture = { has_culture_group = culture_group:skarureh_group }
							culture = { has_culture_group = culture_group:tutelo_group }
						}
					}
					split_pop = {
						fraction = 0.5 #50% of the pop will be changed to the new culture
						culture = culture:elyrics_culture
					}
					#change_pop_culture = culture:elyrics_culture
				}
			}
			every_owned_location = {
				every_pop = {
					limit = {
						culture = culture:elyrics_culture
						pop_type = pop_type:tribesmen
					}
					change_pop_type = pop_type:peasants #slaves
				}
			}
			scope:the_scientist = { move_country = root }
			every_character = {
				change_character_culture = culture:elysian_culture
			}
		}

		culture = {
			add_cultural_tradition = cultural_tradition_extreme_bonus
			add_cultural_influence = cultural_influence_radical_bonus
		}

	}

	option = {
		name = elysium_event.1.b
		historical_option = yes
		high_reward_option = yes

		add_prestige = prestige_extreme_bonus
		
		add_country_modifier = {
			modifier = br_elysium_formed
			years = 10
			mode = add_and_extend
		}
		capital = {
            change_institution_progress = {
                type = institution:meritocracy
                value = 100
             }
        }
		location:anacostia = {
			change_raw_material = goods:silk
		}

		culture = {
			add_cultural_tradition = cultural_tradition_extreme_bonus
			add_cultural_influence = cultural_influence_radical_bonus
		}

	}

}

elysium_event.3 = {
	type = country_event
	title = elysium_event.3.title
	desc = elysium_event.3.desc
	outcome = neutral
	fire_only_once = yes
	illustration_tags = {
		10 = happy
		10 = exterior
	}

	dynamic_historical_event = {
        tag = E1Y
        from = 1337.1.1
        to = 1338.1.1
        monthly_chance = 100
    }
	trigger = {
		tag = E1Y
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate	background = estate_type:clergy_estate }
	}

	option = {
		name = elysium_event.3.a

		hidden_effect = { #Needed so the game knows Hellenism exists.
			add_policy = policy:br_elysian_colonial_settlement
			change_religion = religion:hellenism_religion
			add_religious_aspect = religious_aspect:monism_aspect
			ruler_or_regent ?= {
				change_character_religion = religion:hellenism_religion
				every_child = {
					limit = {
						is_heir = no
					}
					change_character_religion = religion:hellenism_religion
				}
			}
			heir ?= {
				change_character_religion = religion:hellenism_religion
			}
			consort ?= {
				change_character_religion = religion:hellenism_religion
			}
		}
	}
}

br_elysian_DHE.1 = { #Possibility to sponsor Greek Fire
	type = country_event
	title = d008_flavor_byz.1.title
	desc = d008_flavor_byz.1.desc
	outcome = neutral
	fire_only_once = yes

	image = "gfx/interface/illustrations/event/backgrounds/special/greek_fire.dds"

	dynamic_historical_event = {
		tag = IMP
		tag = ELY
		tag = E1Y
		tag = E2Y
		from = 1337.1.1
		to = 1837.1.1
		monthly_chance = 25
	}
	trigger = {
		has_dlc = "d008_fate_of_the_phoenix"
		any_artist = {
			artist_type = scientist
			has_art_in_progress = no
			NOT = { has_character_modifier = busy_modifier }
			artist_skill > 0.7
		}
	}
	immediate = {
		random_artist = {
			limit = {
				artist_type = scientist
				has_art_in_progress = no
				NOT = { has_character_modifier = busy_modifier }
				artist_skill > 0.7
			}
			save_scope_as = target_artist
		}
	}

	option = { #Sponsor
		name = d008_flavor_byz.1.a
		change_gold_effect = { scale = -5 }
		custom_tooltip = {
			text = d008_flavor_byz.1.a.tt
			if = {
				limit = { exists = scope:target_artist }
				scope:target_artist = {
					add_character_modifier = {
						modifier = busy_modifier
						years = -1
						mode = add
					}
					set_variable = searching_for_greek_fire
				}
				random_list = {
					50 = {
						trigger_event_non_silently = { id = d008_flavor_byz.2 days = { 60 180 } }
					}
					50 = {
						trigger_event_non_silently = { id = d008_flavor_byz.3 days = { 60 180 } }
					}
				}
			}
		}
		add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_mild_bonus}
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_mild_bonus}
	}

	option = { #Nah
		name = d008_flavor_byz.1.b
		add_prestige = prestige_mild_bonus
		add_navy_tradition = navy_tradition_mild_bonus
	}
}
