br_ere_disaster_events.4 = { #Border Frictions
	category = disaster_event
    type = country_event
	title = br_ere_disaster_events.4.title
	desc = br_ere_disaster_events.4.desc
	outcome = neutral

	illustration_tags = {
		10 = armed
		10 = exterior
	}

	trigger = {
		any_neighbor_country = {
			NOT = { is_allied_with = { target = ROOT } }
			NOT = { is_subject_or_below_of = root }
			NOT = { has_truce_with = ROOT }
			any_owned_location = {
				any_neighbor_location = {
					owner = ROOT
				}
			}
		}
	}
	image = "gfx/interface/illustrations/disaster/curse_of_stefan_uros_iii.dds"
	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }
		random_neighbor_country = {
			limit = {
				NOT = { is_allied_with = { target = ROOT } }
				NOT = { has_truce_with = ROOT }
				NOT = { root = { has_blocked_treaties = prev } }
				NOT = { is_subject_or_below_of = root }
			}
			save_scope_as = target_country
		}
		random_owned_location = {
			limit = {
				is_neighbor_of = scope:target_country
			}
			province = {
				save_scope_as = border_claim_province
			}
		}
	}

	option = {
		name = br_ere_disaster_events.4.a
		scope:target_country = {
			add_opinion = {
				target = root
				modifier = border_friction_from_event
			}
		}
		scope:target_country = {
			add_casus_belli =  {
				type = casus_belli:cb_conquer_province
				province = scope:border_claim_province
				target = c:BYZ
			}
		}
	}
}

br_ere_disaster_events.6 = { #Province Declares Independence
	category = disaster_event
    type = country_event
	title = br_ere_disaster_events.6.title
	desc = br_ere_disaster_events.6.desc
	outcome = neutral
	image = "gfx/interface/illustrations/disaster/curse_of_stefan_uros_iii.dds"
	trigger = {
		any_owned_location = {
			NOT = {
				any_location_in_province = {
					has_variable = revolter_location
				}
			}
		}
	}

	illustration_tags = {
		10 = armed
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }
		random_owned_location = {
			limit = {
				NOT = {
					any_location_in_province = {
						OR = {
							is_capital = yes
							has_variable = revolter_location
						}
					}
				}
			}
			save_scope_as = target_province
			province_capital = { 
				save_scope_as = target_location 
				set_variable = {
					name = revolter_location
					years = 5
				}
			}
		}
	}

	option = {
		name = br_ere_disaster_events.6.a
		custom_tooltip = {
			text = people_of_target_province_will_declare_war_on_us_tt
					
			scope:target_province = {
				create_location_country_from_province = {
					save_scope_as = target_country
				}
			}
			c:BYZ ?= {
				declare_war_with_cb = {
					target = scope:target_country
					type = casus_belli:cb_annex
				}
			}
		}
	}

	option = {
		name = br_ere_disaster_events.6.b
		scope:target_province = {
			every_location_in_province = { change_control = self_control_severe_penalty }
		}
		scope:target_location = {
			ordered_pop = {
				limit = { owner = root }
				order_by = pop_size
				max = 3
				check_range_bounds = no
				add_pop_satisfaction = pop_satisfaction_mild_penalty
			}
		}
		add_stability = stability_weak_penalty
	}
}

br_ere_disaster_events.12 = { #Feuding Estates
	category = disaster_event
	type = country_event
	title = br_ere_disaster_events.12.title
	desc = br_ere_disaster_events.12.desc
	outcome = neutral
	image = "gfx/interface/illustrations/disaster/twilight_of_the_tsardom.dds"

	trigger = {
		any_estate = {
			NOT = { estate_type = estate_type:crown_estate }
			count >= 2
		}
	}

	illustration_tags = {
        10 = angry
        10 = interior
    }

	immediate = {
		random_estate = {
			limit = {
				NOT = { estate_type = estate_type:crown_estate }
			}
			save_scope_as = first_estate
		}
		random_estate = {
			limit = {
				NOT = { estate_type = estate_type:crown_estate }
				this != scope:first_estate
			}
			save_scope_as = second_estate
		}
	}
	
	option = {
		name = br_ere_disaster_events.12.a
		add_estate_satisfaction = { type = scope:first_estate.estate_type value = estate_satisfaction_mild_bonus }
		add_estate_satisfaction = { type = scope:second_estate.estate_type value = estate_satisfaction_mild_penalty }
	}
	
	option = {
		name = br_ere_disaster_events.12.b
		add_estate_satisfaction = { type = scope:first_estate.estate_type value = estate_satisfaction_mild_penalty }
		add_estate_satisfaction = { type = scope:second_estate.estate_type value = estate_satisfaction_mild_bonus }
	}
	
	option = {
		name = br_ere_disaster_events.12.c
		add_prestige = prestige_mild_bonus
		add_estate_satisfaction = { type = scope:first_estate.estate_type value = estate_satisfaction_severe_penalty }
