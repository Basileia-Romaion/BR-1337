# Catalan Crusade Announced
br_latin_wars.1 = {
	hide_portraits = yes
	type = country_event
	title = br_latin_wars.1.title
	desc = br_latin_wars.1.desc
	outcome = neutral

	fire_only_once = yes
	dynamic_historical_event = {
		tag = PAP
		from = 1337.1.1
		to = 1370.1.1
		monthly_chance = 10
	}

	trigger = {
		location:zaragoza = {
			owner.religion.group != ROOT.religion.group
		}
		religion = religion:catholic
		country_exists = c:ARA
		crusade_potential_trigger = yes
		crusade_allow_trigger = yes
	}

	illustration_tags = {
        10 = angry
        10 = exterior
    }

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:clergy_estate }
		location:zaragoza.owner = {
			save_scope_as = crusade_target
		}
		c:ARA = {
			if = {
				limit = { religion = religion:catholic }
				add_to_list = list_of_nations_to_invite_to_crusade
			}
		}
		c:KNI ?= {
			if = {
				limit = { religion = religion:catholic }
				add_to_list = list_of_nations_to_invite_to_crusade
			}
		}
		c:TOU ?= {
			if = {
				limit = { religion = religion:catholic }
				add_to_list = list_of_nations_to_invite_to_crusade
			}
		}
		c:PRO ?= {
			if = {
				limit = { religion = religion:catholic }
				add_to_list = list_of_nations_to_invite_to_crusade
			}
		}
		save_scope_as = papal_state
	}

	option = {
		name = br_latin_wars.1.a
		historical_option = yes
		set_global_variable = {
			name = crusade_location_target
			value = location:zaragoza
		}
		crusade_effect = { target = scope:crusade_target }

		c:ARA = {
			set_variable = receive_additional_benefits
		}

		hidden_effect = {
			scope:crusade_target = {
				random_current_war = {
					limit = {
						OR = {
							attacker_leader = ROOT
							defender_leader = ROOT
						}
					}
					save_scope_as = crusade_war
				}
			}
		}

		every_in_list = {
			list = list_of_nations_to_invite_to_crusade
			trigger_event_non_silently = br_latin_wars.2
		}
		scope:crusade_target = {
			trigger_event_non_silently = br_latin_wars.3
		}
		hidden_effect = {
			scope:crusade_war = {
				set_variable = is_smyrniote_crusade
			}
		}
	}

	option = {
		name = br_latin_wars.1.b
	
		add_prestige = prestige_severe_penalty
	}
}

# Catalan Crusade - Call to Arms
br_latin_wars.2 = {
	type = country_event
	title = br_latin_wars.2.title
	desc = br_latin_wars.2.desc
	outcome = neutral

	illustration_tags = {
        10 = angry
        10 = exterior
    }

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:clergy_estate }
		if = {
			limit = {
				has_variable = receive_additional_benefits
			}
			set_local_variable = {
				name = nobles_going_to_war
				value = 0
			}
			religion:catholic = {
				every_country_in_religion = {
					limit = {
						capital = { continent = continent:europe }
					}
					every_pop = {
						limit = {
							religion = religion:catholic
							pop_type = pop_type:nobles
						}
						change_local_variable = {
							name = nobles_going_to_war
							add = pop_size
						}
					}		
				}
			}
			change_local_variable = {
				name = nobles_going_to_war
				multiply = 0.02
			}
		}
	}

	option = {
		name = br_latin_wars.2.a

		historical_option = yes
		
		join_war_as_attacker = { war = scope:crusade_war }
		if = {
			limit = {
				has_variable = receive_additional_benefits
			}
			remove_variable = receive_additional_benefits
			add_manpower = local_var:nobles_going_to_war
			add_gold = {
				value = local_var:nobles_going_to_war
				multiply = 5
			}
		}
		add_country_modifier = {
			modifier = kni_smyrniote_crusade
			years = 5
		}
		add_opinion_mutual_effect = {
			target = c:PAP
			modifier = kni_joined_crusade
		}
		if = {
			limit = {
				this = c:ARA
			}
			set_variable = participated_in_smyrniote_crusade
		}
	}

	option = {
		name = br_latin_wars.2.b
	
		add_prestige = prestige_severe_penalty
	}
}
