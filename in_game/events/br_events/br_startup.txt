br_startup.2 = {
	type = country_event
	title = empty_text
	desc = empty_text
	hidden = yes
	outcome = neutral
	trigger = {
		tag = GRA
	}

	fire_only_once = yes

	immediate = {
		change_tag_cosmetic = { tag = ADU } #So it has the correct name until the tag is formed.
		if = {
			limit = { country_rank_level < 4 }
			set_country_rank_effect = { rank = country_rank:rank_empire }
		}
		set_capital = location:cordoba
	}

	option = {
		c:GRA ?= {
			form_country = formable_country:ADU_f_GRA
		}
	}
}
