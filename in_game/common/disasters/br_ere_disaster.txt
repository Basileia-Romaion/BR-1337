##disaster created from various diasters but primarily used curse_of_stefan_uros_iii as the base with the goal of having regular mid-late game challenge for the ere.

br_ere_disaster = {
	image = "gfx/interface/illustrations/disaster/byzantine_succ_crisis.dds"
	monthly_spawn_chance = monthly_spawn_chance_very_high
	#fire_only_once = yes

	can_start = {
		OR = {
			tag = BYZ
			tag = ELY
			tag = IMP
		}
		NOT = { 
			had_disaster_trigger = {
				type = br_ere_disaster
			}
		}

		has_any_active_disaster = no
		current_age_or_later = { age = age_3_discovery }
		num_locations > 250

		ruler ?= { total_abilities <= 250 }
		stability <= 30
		government_power <= 75

	}

	can_end = {
		br_ere_disaster_end_trigger = yes
	}
	modifier = {
		monthly_pretender_rebel_growth = 0.01
		monthly_nationalist_rebel_growth = 0.01
		monthly_legitimacy = -0.25
		stability_cost_efficiency = -0.25
	}

	on_start = {
		trigger_event_non_silently = br_ere_disaster_events.1
	}
	on_end = {
		trigger_event_non_silently = br_ere_disaster_events.2
	}

	on_monthly = {
		random_list = {
			20 = { custom_tooltip = an_event_occurs_tt }
			80 = { custom_tooltip = no_event_occurs_tt }
		}
		hidden_effect = {
			random_list = {
				80 = {}
				4 = { trigger_event_silently = br_ere_disaster_events.3 }
				4 = { trigger_event_silently = br_ere_disaster_events.4 }
				4 = { trigger_event_silently = br_ere_disaster_events.5 }
				2 = { trigger_event_silently = br_ere_disaster_events.6 }
				5 = { trigger_event_silently = br_ere_disaster_events.7 }
				4 = { trigger_event_silently = br_ere_disaster_events.8 }
				4 = { trigger_event_silently = br_ere_disaster_events.9 }
				4 = { trigger_event_silently = br_ere_disaster_events.10 }
				4 = { trigger_event_silently = br_ere_disaster_events.11 }
				2 = { trigger_event_silently = br_ere_disaster_events.12 }
				2 = { trigger_event_silently = br_ere_disaster_events.13 }
			}
		}
	}
}
