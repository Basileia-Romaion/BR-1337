change_roman_authority = {
	international_organization:roman_world ?= {
		custom_description = {
			text = change_roman_authority_text
			value = $value$
			change_variable = {
				name = roman_authority
				add = $value$
			}
			if = {
				limit = { has_variable = roman_authority }
				if = {
					limit = { var:roman_authority > 100 }
					set_variable = { name = roman_authority value = 100 }
				}
				if = {
					limit = { var:roman_authority < 0 }
					set_variable = { name = roman_authority value = 0 }
				}
			}
		}
	}
}

add_decree_monthly_roman_authority_loss = {
	add_country_modifier = {
		modifier = roman_decree_maintenance
		years = 5
		mode = add_and_extend
	}
}

remove_decree_monthly_roman_authority_loss = {
	change_country_modifier_size = {
		modifier = roman_decree_maintenance
		value = -1
	}
	if = {
		limit = { country_modifier_strength = { modifier = roman_decree_maintenance value <= 0 } }
		remove_country_modifier = roman_decree_maintenance
	}
}
