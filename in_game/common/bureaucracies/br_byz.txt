	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_dlc = "d008_fate_of_the_phoenix"
		OR = {
			has_or_had_tag = BYZ
			has_or_had_tag = TRE
			culture = { has_culture_group = culture_group:roman_group } #br added
		}
	}

	allow = {
		OR = {
			NOT = {
				any_current_bureaucracy = {
					bureaucracy_type = bureaucracy_type:magister_militum_bureaucracy
				}
			}
			has_reform = government_reform:theme_system
		}
	}

	estates_that_like = {
		nobles_estate
	}
	estates_that_dislike = {
		burghers_estate
	}

	#neutral
	neutral_modifier = {
		global_manpower_modifier = 0.05
		global_levy_size_modifier = 0.05
		monthly_towards_stratiotai = societal_value_monthly_move #br added
	}

	#positive
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_defensive = 0.2
		monthly_roman_authority = 0.02
	}

	#negative
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		levy_recovery_modifier = -0.2
		global_nobles_estate_power = 0.5
		monthly_roman_authority = -0.02
	}
}
