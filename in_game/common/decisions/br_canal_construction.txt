
build_niagara_canal = {
	decision_category = infrastructure_decisions
	hide_repeatable_tag = yes

	potential = {
		has_discovered = location:niagara
	}

	allow = {
		gold >= 5000
		owns_or_non_sovereign_subject_owns = location:niagara
		has_advance = niagara_canal_advance
		location:niagara = {
			is_canal_open = no
			NOT = { has_canal_construction = yes }
		}
	}

	ai_will_do = {
		value = -1
		if = {
			limit = {
				gold >= 10000
				last_months_balance > 0
			}
			add = 100
		}
	}

	option = {
		name = build_niagara_canal.a
		ai_chance = { value = 100 }
		effect = {
			location:niagara = {
				start_canal_construction = { }			
			}
			add_gold = -5000
		}
	}
}
