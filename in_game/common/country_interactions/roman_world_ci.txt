#Domestikos of the West
roman_world_bestow_domestikos_west = {
	type = diplomacy
	category = CATEGORY_HRE_ACTIONS

	potential = {
		exists = international_organization:roman_world
		international_organization:roman_world = {
			leader_country ?= scope:actor
			"num_countries_with_special_status(special_status:domestikos_west)" < "max_countries_with_special_status(special_status:domestikos_west)"
		}
	}

	allow = {
	}
	
	select_trigger = {
		looking_for_a = country
		target_flag = recipient
		name = "choose_country"
		column = {
			data = name
		}
		visible = {
			is_member_of_international_organization = international_organization:roman_world
		}
        enabled = {
			international_organization:roman_world = {
				special_status_can_be_bestowed = {
					type = special_status:domestikos_west
					country = root
				}
			}
		}
	}

	effect = {
		scope:actor = {
			change_roman_authority = { value = -5 }
            add_legitimacy = -10 #added as a cost
		}
		international_organization:roman_world = {
			international_organization_add_special_status = {
				type = special_status:domestikos_west
				country = scope:recipient
			}
			international_organization_remove_special_status = {
				type = special_status:imperial_loyalist
				country = scope:recipient
			}
		}
	}

	ai_will_do = {
		value = -100
		add = {
			desc = "OUR_OPINION_TOOLTIP"
			value = "scope:actor.opinion(scope:recipient)"
		}
		add = {
			desc = "THEIR_OPINION_TOOLTIP"
			value = "scope:recipient.opinion(scope:actor)"
		}
		if = {
			limit = {
				scope:actor.culture = scope:recipient.culture
			}
			add = 50
		}
		else_if = {
			limit = {
				scope:actor = {
					any_accepted_culture = {
						this = scope:recipient.culture
					}
				}
			}
			add = 25
		}
	}

	accept = {
		if = {
			limit = { exists = scope:recipient }
			add = {
				desc = "THEIR_OPINION_TOOLTIP"
				value = "scope:recipient.opinion(scope:actor)"
				if = {
					limit = {
						scope:recipient = {
							"opinion(scope:actor)" > 0
						}
					}
					divide = 10
				}
				else = {
					divide = 5
				}
			}
			add = {
				scope:actor = {
					value = modifier:diplomatic_reputation
					multiply = 3
				}
			}
			add = {
				scope:recipient = {
					value = "trust(scope:actor)"
					divide = 5
				}
			}
		}
	}
}

# Subjugate
br_restore_roman_status = {
	type = diplomacy
	category = CATEGORY_HRE_ACTIONS

	ai_limit_per_check = 1

	potential = {
		exists = international_organization:roman_world
		international_organization:roman_world = {
			leader_country ?= scope:actor
		}
	}

	allow = {
		scope:actor = {
            is_leader_of_international_organization = international_organization:roman_world
		}
    }
	
	select_trigger = {
		looking_for_a = country
		target_flag = recipient
		name = "choose_country"
		column = {
			data = name
		}
		visible = {
			is_member_of_international_organization = international_organization:roman_world
			is_rebel_country = no
			international_organization:roman_world = {
				country_has_special_status = {
					type = special_status:defiant_strategoi
					country = root
				}
			}
		}
        enabled = {
			has_special_status_in_international_organization = {
                type = special_status:defiant_strategoi
                international_organization = international_organization:roman_world
            }
		}
	}

	effect = {
		scope:actor = {
            change_roman_authority = { value = -10 }
            add_legitimacy = -10 #added as a cost
			add_prestige = 5
		}
		scope:recipient = {
			make_subject_of = {
				target = scope:actor
				type = subject_type:br_thema_subject
			}
		}
	}

	ai_will_do = {
		add = {
			desc = "OUR_OPINION_TOOLTIP"
			value = "scope:actor.opinion(scope:recipient)"
		}
		add = {
			desc = "THEIR_OPINION_TOOLTIP"
			value = "scope:recipient.opinion(scope:actor)"
		}
		if = {
			limit = { scope:actor.culture.language = scope:recipient.culture.language }
			add = {
				desc = "UPKEEP_SAME_LANGUAGE"
				value = 25
			}
		}
		if = {
			limit = {
				not = {
					scope:recipient = {
						government_type = government_type:republic
					}
				}
			}
			divide = 2
		}
		if = {
			limit = {
				exists = scope:recipient
				scope:recipient = { has_cooldown = ai_spam_protection }
			}
			subtract = {
				desc = "DIPLOREASON_ALREADY_DID_ACTION"
				value = 10000
			}
		}
	}

	accept = {
		subtract = {
        	desc = "VOTE_BASE_VALUE"
       		value = 30
   		}
    	if = {
        	limit = { scope:actor = { is_allied_with = { target = scope:recipient } } }
        	add = {
            	desc = "game_concept_alliance"
            	value = 15
        	}
    	}
    	if = {
        	limit = { scope:recipient = { is_rival_of = scope:actor } }
        	subtract = {
            	desc = "game_concept_rival"
            	value = 1000
        	}
    	}
    	if = {
        	limit = { scope:recipient = { is_enemy_of = scope:actor } }
        	subtract = {
            	desc = "game_concept_enemy"
            	value = 50
        	}
    	}
    	add = {
        	desc = "THEIR_OPINION_TOOLTIP"
        	value = "scope:recipient.opinion(scope:actor)"
        	divide = 10
    	}
   	 	add = {
        	desc = "IO_LEADER_DIP_REP"
        	scope:actor = {
        	    value = modifier:diplomatic_reputation
        	    #multiply = 2
        	}
    	}
    	add = {
       	 	desc = "game_concept_trust"
			scope:recipient = {
				value = "trust(scope:actor)"
				divide = 10
			}
    	}
		add = {
			desc = "DISTANCE_FROM_ROMANLAND"
			value = "scope:recipient.border_distance_to(scope:actor)"
			multiply = -0.05
			min = -1000
		}
	}

	reject_effect = {
		hidden_effect = {
			scope:recipient = {
				add_cooldown = {
					type = ai_spam_protection
					years = 5
				}
			}
		}
	}
}

#These are for the Theme subject type to allow changing ruler and allowing or disallowng dynastic inheritance or direct imperial inheritance on stategos death.
br_change_strategos = {
	type = diplomacy
	category = CATEGORY_HRE_ACTIONS
	ai_tick = never
	ai_tick_frequency = 120

	potential = {
		exists = international_organization:roman_world
		international_organization:roman_world = {
			leader_country ?= scope:actor
		}
	}

	allow = {
		#scope:actor = {
			#exists = ruler_or_heir_if_regent.dynasty
			#ruler_or_heir_if_regent.dynasty = {
				#num_adult_available_dynasty_members > 0
			#}
		#}
	}
	
	select_trigger = {
		looking_for_a = country
		target_flag = recipient
		interaction_source_list = {
			scope:actor = {
				every_subject = {
					add_to_list = source
				}
			}
		}

		cache_targets = yes
		target_flag = recipient
		name = "choose_country"
		column = {
			data = name
		}

		visible = {
			is_subject_type = br_thema_subject
			is_subject_of = scope:actor
		}
		enabled = {
			#has_regent = yes
			#has_heir = no
			not = { is_at_war_with = scope:actor }
			#subject_loyalty >= MIN_SUBJECT_LOYALTY_FOR_ACTION
			#liberty_desire < MAX_SUBJECT_LIBERTY_DESIRE_FOR_ACTION
		}
	}

	select_trigger = {
		looking_for_a = character
		source = actor
		#target_flag = target
		ai_interaction_source_list = {
			scope:actor = {
				ruler_or_heir_if_regent.dynasty = {
					every_available_dynasty_member = {
						limit = {
							religion = scope:actor.religion
							modifier:blocked_from_character_interactions = no
						}
						add_to_list = source
					}
				}
			}
		}

		cache_interaction_source_list = yes
		target_flag = target
		name = "change_ruler_SELECT_TITLE"
		none_available_msg_key = "change_ruler_no_characters"
		column = {
			data = character_ruler
		}

		visible = {
			is_alive = yes
			is_ruler = no
			is_heir = no
			has_dynasty = yes
			#dynasty = scope:actor.ruler_or_heir_if_regent.dynasty
			is_adult = yes
			religion = scope:actor.religion
			modifier:blocked_from_character_interactions = no
		}
	}
	
	effect = {
		scope:actor = {
			change_roman_authority = { value = -5 }
            add_legitimacy = -10 #added as a cost
		}
		if = {
			limit = { exists = scope:target }
			scope:target = { move_country = scope:recipient }
			scope:recipient = { set_new_ruler = scope:target }
		}
	}
	
	block_when_at_war = yes
	
	ai_will_do = {
		add = -10
	}
}
