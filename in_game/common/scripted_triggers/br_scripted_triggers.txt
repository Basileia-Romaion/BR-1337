### Roman - Unit Triggers ###

REPLACE_OR_CREATE:unit_akritai_location_trigger = {
	dominant_culture ?= { has_culture_group = culture_group:roman_group }
	owner ?= {
		culture = { has_culture_group = culture_group:roman_group }
		has_advance = byz_akritai_advance
	}
}

		culture = { has_culture_group = culture_group:roman_group }

	}
}

br_ere_disaster_end_trigger = {
	OR = {
		ruler ?= { total_abilities >= 250 }

		AND = {
			stability >= 45
			government_power >= 75
		}

		NOT = { government_type = government_type:monarchy }
		in_civil_war = yes

		had_disaster_for_years = {
			disaster_type = disaster_type:br_ere_disaster
			years >= 25
		}
	}
}
