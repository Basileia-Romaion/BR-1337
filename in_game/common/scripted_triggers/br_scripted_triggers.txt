### Roman - Unit Triggers ###

REPLACE_OR_CREATE:unit_akritai_location_trigger = {
	dominant_culture ?= { has_culture_group = culture_group:greek_group }
	owner ?= {
		culture = { has_culture_group = culture_group:greek_group }
		has_advance = byz_akritai_advance
	}
}

	owner ?= {
		culture = { has_culture_group = culture_group:greek_group }
		has_advance = byz_cataphracts_advance
	}
}
